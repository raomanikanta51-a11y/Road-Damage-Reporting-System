function escapeHtml(value) {
  return String(value ?? "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/\"/g, "&quot;")
    .replace(/'/g, "&#039;");
}

function readStorage(key, fallback) {
  if (typeof localStorage === "undefined") return fallback;
  try {
    const value = localStorage.getItem(key);
    if (!value) return fallback;
    return JSON.parse(value);
  } catch (error) {
    return fallback;
  }
}

function writeStorage(key, value) {
  if (typeof localStorage === "undefined") return;
  if (value === null) {
    localStorage.removeItem(key);
    return;
  }
  localStorage.setItem(key, JSON.stringify(value));
}

function getReports() {
  const stored = readStorage(RKEY, null);
  if (Array.isArray(stored) && stored.length) return stored;
  writeStorage(RKEY, seedReports);
  return [...seedReports];
}

function setReports(nextReports) {
  writeStorage(RKEY, nextReports);
}

function getUser() {
  return readStorage(UKEY, DEFAULT_USER);
}

function saveUser(user) {
  writeStorage(UKEY, user);
}

function getRoute() {
  const hash = window.location.hash.replace(/^#/, "").trim();
  if (!hash || hash === "/") return "home";
  const normalized = hash.replace(/^\//, "");
  return normalized || "home";
}

function formatDate(value) {
  return new Date(value).toLocaleDateString();
}

function getSummaryStats(reports) {
  return [
    { label: "Total reports", value: reports.length },
    { label: "In progress", value: reports.filter((report) => report.status === "in-progress").length },
    { label: "Resolved", value: reports.filter((report) => report.status === "resolved").length },
    { label: "High severity", value: reports.filter((report) => report.severity === "high").length },
  ];
}

function renderNav(user) {
  const route = getRoute();
  const menu = [
    { route: "home", label: "Home" },
    { route: "reports", label: "Reports" },
    { route: "report", label: "Report Damage" },
    { route: "about", label: "Project" },
  ];

  const links = menu
    .map(({ route: target, label }) => {
      const isActive = target === route ? "active" : "";
      return `<a href="#/${target === "home" ? "" : target}" class="nav-link ${isActive}" data-route="${target}">${escapeHtml(label)}</a>`;
    })
    .join("");

  const authBlock = user
    ? `<button type="button" class="user-pill" data-action="logout">${escapeHtml(user.name)} · ${escapeHtml(user.role)} ✕</button>`
    : `<a href="#/login" class="primary-btn" data-route="login">Sign in</a>`;

  return `
    <header class="nav" aria-label="Main navigation">
      <a href="#/" class="brand" data-route="home"><span class="grad">#1</span> RoadWatch</a>
      ${links}
      ${authBlock}
    </header>
  `;
}

              `,
            )
            .join("")}
        </div>
      </div>

      <div class="report-grid">
        ${
          filtered.length
            ? filtered
                .map((report, index) => {
                  const canEdit = !!user && (user.role === "admin" || user.name === report.reporter);
                  const imageBlock = report.image
                    ? `<img class="report-image" src="${report.image}" alt="${escapeHtml(report.damageType)}" />`
                    : `<div class="report-image" style="display:flex; align-items:center; justify-content:center; font-size:2.6rem; font-weight:900; background:linear-gradient(135deg, rgba(56,189,248,0.12), rgba(251,191,36,0.08)); color:var(--primary);">#${String(index + 1).padStart(2, "0")}</div>`;

                  const severityClass = `severity-${report.severity}`;
                  const statusClass = `status-${report.status}`;
                  const editControls = state.editingId === report.id
                    ? `
                      <form data-form="report-edit" style="display:grid; gap:10px; margin-top:12px;">
                        <input type="hidden" name="reportId" value="${report.id}" />
                        <input class="field" type="text" name="location" value="${escapeHtml(report.location)}" required />
                        <textarea class="textarea" name="description" required>${escapeHtml(report.description)}</textarea>
                        <button type="submit" class="primary-btn">Save</button>
                      </form>
                    `
                    : `
                      <div class="report-details">
                        <div>📍 ${escapeHtml(report.location)}</div>
                        <p class="muted" style="margin:0;">${escapeHtml(report.description)}</p>
                      </div>
                    `;

                  return `
                    <article class="report-card">
                      ${imageBlock}
                      <div class="report-body">
                        <div class="report-head">
                          <span class="report-tag">${escapeHtml(report.damageType)}</span>
                          <span class="status-tag ${statusClass}">${escapeHtml(report.status)}</span>
                        </div>
                        ${editControls}
                        <div class="meta-row">
                          <span class="severity-tag ${severityClass}">${escapeHtml(report.severity)}</span>
                          <span>${escapeHtml(report.reporter)} · ${formatDate(report.createdAt)}</span>
                        </div>
                        ${canEdit ? `
                          <div class="card-actions">
                            ${user && user.role === "admin" ? `
                              <select class="select" data-status-update="${report.id}" style="max-width:170px;">
                                <option value="reported" ${report.status === "reported" ? "selected" : ""}>Reported</option>
                                <option value="in-progress" ${report.status === "in-progress" ? "selected" : ""}>In progress</option>
                                <option value="resolved" ${report.status === "resolved" ? "selected" : ""}>Resolved</option>
                              </select>
                            ` : ""}
                            <button type="button" class="secondary-btn" data-action="toggle-edit" data-id="${report.id}">${state.editingId === report.id ? "Cancel" : "Edit"}</button>
                            ${user && user.role === "admin" ? `<button type="button" class="danger-btn" data-action="delete-report" data-id="${report.id}">Delete</button>` : ""}
                          </div>
                        ` : ""}
                      </div>
                    </article>
                  `;
                })
                .join("")
            : `<div class="empty-state" style="grid-column:1 / -1;">No reports match.</div>`
        }
      </div>
    </main>
  `;
}

function renderContent() {
  const user = getUser();
  const reports = getReports();

  if (activeRoute === "login") {
    appRoot.innerHTML = renderLoginPage(user);
    return;
  }

  if (activeRoute === "report") {
    appRoot.innerHTML = renderReportPage(user);
    return;
  }

  if (activeRoute === "reports") {
    appRoot.innerHTML = renderReportsPage(user);
    return;
  }

  if (activeRoute === "about") {
    appRoot.innerHTML = renderAboutPage(user);
    return;
  }

  appRoot.innerHTML = renderHomePage(reports, user);
}

function boot() {
  if (!document || !document.body) {
    return;
  }

  if (!document.getElementById(appRootId)) {
    appRoot = document.createElement("div");
    appRoot.id = appRootId;
    document.body.appendChild(appRoot);
  } else {
    appRoot = document.getElementById(appRootId);
  }

  if (!document.getElementById("roadwatch-styles")) {
    const cssNode = document.createElement("style");
    cssNode.id = "roadwatch-styles";
    cssNode.textContent = themeStyles;
    document.head.appendChild(cssNode);
  }

  activeRoute = getRoute();
  renderContent();
}

document.addEventListener("click", (event) => {
  const navLink = event.target.closest("[data-route]");
  if (navLink) {
    event.preventDefault();
    const nextRoute = navLink.getAttribute("data-route");
    activeRoute = nextRoute || "home";
    if (nextRoute === "home") {
      window.location.hash = "#/";
    } else {
      window.location.hash = `#/${nextRoute}`;
    }
    renderContent();
    return;
  }

  const statusFilter = event.target.closest("[data-status-filter]");
  if (statusFilter) {
    state.status = statusFilter.getAttribute("data-status-filter") || "all";
    renderContent();
    return;
  }

  const toggleEdit = event.target.closest("[data-action='toggle-edit']");
  if (toggleEdit) {
    const reportId = toggleEdit.getAttribute("data-id");
    state.editingId = state.editingId === reportId ? null : reportId;
    renderContent();
    return;
  }

  const deleteButton = event.target.closest("[data-action='delete-report']");
  if (deleteButton) {
    const reportId = deleteButton.getAttribute("data-id");
    const reports = getReports();
    const nextReports = reports.filter((report) => report.id !== reportId);
    setReports(nextReports);
    renderContent();
    return;
  }

  const logoutButton = event.target.closest("[data-action='logout']");
  if (logoutButton) {
    saveUser(null);
    renderContent();
  }

  const locateButton = event.target.closest("[data-action='locate-gps']");
  if (locateButton && navigator && navigator.geolocation) {
    navigator.geolocation.getCurrentPosition((position) => {
      const locationField = document.querySelector("[name='location']");
      if (!locationField) return;
      const value = `${position.coords.latitude.toFixed(5)}, ${position.coords.longitude.toFixed(5)}`;
      locationField.value = value;
    });
  }
});

document.addEventListener("input", (event) => {
  const searchField = event.target.closest("[data-search-input]");
  if (searchField) {
    state.search = searchField.value || "";
    renderContent();
  }

  const radio = event.target.closest("input[type='radio'][name='role']");
  if (radio) {
    const parent = radio.closest(".toggle");
    if (parent) {
      document.querySelectorAll(".toggle").forEach((toggle) => toggle.classList.toggle("active", toggle.contains(radio)));
    }
  }
});

document.addEventListener("change", (event) => {
  const reportStatus = event.target.closest("[data-status-update]");
  if (reportStatus) {
    const reportId = reportStatus.getAttribute("data-status-update");
    const value = reportStatus.value;
    const reports = getReports();
    const nextReports = reports.map((report) => (report.id === reportId ? { ...report, status: value } : report));
    setReports(nextReports);
    renderContent();
  }

  const fileInput = event.target.closest("input[type='file']");
  if (fileInput && fileInput.files && fileInput.files[0]) {
    const file = fileInput.files[0];
    const reader = new FileReader();
    reader.onload = () => {
      const uploadBox = fileInput.closest(".upload-box");
      if (!uploadBox) return;
      const preview = uploadBox.querySelector("img");
      if (preview) {
        preview.src = String(reader.result);
      } else {
        const image = document.createElement("img");
        image.src = String(reader.result);
        uploadBox.insertBefore(image, uploadBox.firstChild);
      }
    };
    reader.readAsDataURL(file);
  }
});

document.addEventListener("submit", (event) => {
  const form = event.target;

  if (form.matches("[data-form='login']")) {
    event.preventDefault();
    const formData = new FormData(form);
    const name = String(formData.get("name") || "").trim();
    const role = String(formData.get("role") || "citizen");
    if (!name) return;
    saveUser({ name, role });
    activeRoute = "reports";
    window.location.hash = "#/reports";
    renderContent();
    return;
  }

  if (form.matches("[data-form='report-create']")) {
    event.preventDefault();
    const formData = new FormData(form);
    const user = getUser();
    const location = String(formData.get("location") || "").trim();
    const description = String(formData.get("description") || "").trim();
    if (!user || !location || !description) return;

    const rawImage = formData.get("photo");
    const reports = getReports();
    const newReport = {
      id: `r_${Date.now()}_${Math.random().toString(16).slice(2)}`,
      location,
      damageType: String(formData.get("damageType") || "Pothole"),
      severity: String(formData.get("severity") || "medium"),
      description,
      image: undefined,
      status: "reported",
      reporter: user.name,
      createdAt: Date.now(),
    };

    if (rawImage && rawImage instanceof File && rawImage.size > 0) {
      const reader = new FileReader();
      reader.onload = () => {
        newReport.image = String(reader.result);
        setReports([newReport, ...reports]);
        activeRoute = "reports";
        window.location.hash = "#/reports";
        renderContent();
      };
      reader.readAsDataURL(rawImage);
      return;
    }

    setReports([newReport, ...reports]);
    activeRoute = "reports";
    window.location.hash = "#/reports";
    renderContent();
    return;
  }

  if (form.matches("[data-form='report-edit']")) {
    event.preventDefault();
    const formData = new FormData(form);
    const reportId = String(formData.get("reportId") || "");
    const location = String(formData.get("location") || "").trim();
    const description = String(formData.get("description") || "").trim();
    if (!reportId || !location || !description) return;

    const reports = getReports();
    const nextReports = reports.map((report) => {
      if (report.id !== reportId) return report;
      return { ...report, location, description };
    });

    setReports(nextReports);
    state.editingId = null;
    renderContent();
  }
});

window.addEventListener("hashchange", () => {
  activeRoute = getRoute();
  renderContent();
});

boot();
